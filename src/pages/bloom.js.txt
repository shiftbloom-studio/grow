// The Bloom: drawn from the brandbook v2 construction spec (rings 10/10/8/6 + seed, 36° symmetry).
(() => {
  const RINGS = [
    { d: 200, n: 10, from: "#D8163F", to: "#FF2E52" },
    { d: 148, n: 10, from: "#FF2E52", to: "#FF7E9E" },
    { d: 100, n: 8, from: "#FF7E9E", to: "#FFC2D3" },
    { d: 54, n: 6, from: "#FFB4CA", to: "#FFDCE7" },
  ];
  let uid = 0;
  function petal(r, n) {
    const w = r * Math.tan(Math.PI / n) * 0.95;
    const f = (x) => x.toFixed(2);
    return `M0 0C${f(w)} ${f(-r * 0.3)} ${f(w * 0.55)} ${f(-r * 0.78)} 0 ${f(-r)}C${f(-w * 0.55)} ${f(-r * 0.78)} ${f(-w)} ${f(-r * 0.3)} 0 0Z`;
  }
  function draw(el) {
    const id = "b" + uid++;
    const solid = el.dataset.solid;
    let defs = "", body = "";
    RINGS.forEach((ring, i) => {
      const r = ring.d / 2;
      const gid = `${id}g${i}`;
      defs += `<linearGradient id="${gid}" gradientUnits="userSpaceOnUse" x1="0" y1="0" x2="0" y2="${-r}"><stop offset="0" stop-color="${ring.from}"/><stop offset="1" stop-color="${ring.to}"/></linearGradient>`;
      const p = petal(r, ring.n);
      const off = i % 2 ? 180 / ring.n : 0;
      let petals = "";
      for (let k = 0; k < ring.n; k++) petals += `<path d="${p}" transform="rotate(${off + (360 / ring.n) * k})" fill="${solid || `url(#${gid})`}"/>`;
      body += `<g class="ring ring${i}">${petals}</g>`;
    });
    body += `<circle class="seed" r="7" fill="${solid || "#FF2E52"}"/>`;
    el.innerHTML = `<svg viewBox="-104 -104 208 208" role="img" aria-label="shiftbloom bloom"><defs>${defs}</defs><g class="bloom-root">${body}</g></svg>`;
  }
  window.drawBlooms = () => document.querySelectorAll("[data-bloom]").forEach(draw);
  if (document.readyState !== "loading") window.drawBlooms();
  else document.addEventListener("DOMContentLoaded", window.drawBlooms);
})();
